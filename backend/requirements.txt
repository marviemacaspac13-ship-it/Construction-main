# TRACE backend - everything needed to run the API and its tests.
#
# Requires Python 3.13 (the version these pins were tested on; 393 tests
# pass). Other Python versions may lack pre-built wheels for opencv, numpy
# or onnxruntime and fail to install.
#
#   cd backend
#   python -m venv .venv
#   .venv\Scripts\python.exe -m pip install -r requirements.txt     (Windows)
#   .venv/bin/python -m pip install -r requirements.txt              (macOS / Linux)
#
# Versions are pinned exactly to a known-working set, so a fresh machine
# installs what was actually tested rather than whatever is newest.
# Verified by installing into an empty environment and running the full
# suite: 393 passed.
#
# WINDOWS - if pip fails with "No such file or directory ... numpy\f2py\..."
# and a hint about "Long Path support", the project folder is too deep.
# numpy ships files nested so deeply that the full path can pass Windows'
# 260-character limit. Either move the project to a short folder such as
# C:\trace, or enable long paths once (PowerShell as Administrator):
#   New-ItemProperty -Path "HKLM:\SYSTEM\CurrentControlSet\Control\FileSystem" `
#     -Name "LongPathsEnabled" -Value 1 -PropertyType DWORD -Force
# then delete the half-built .venv and install again.
